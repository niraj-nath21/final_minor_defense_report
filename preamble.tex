% ---------- Chapter / section heading style ----------
\usepackage{titlesec}
% "CHAPTER N" on one line, TITLE centred beneath it
\titleformat{\chapter}[display]
  {\normalfont\Large\bfseries\centering}
  {CHAPTER \thechapter}{12pt}{\Large\MakeUppercase}
\titlespacing*{\chapter}{0pt}{-20pt}{28pt}

\titleformat{\section}{\normalfont\large\bfseries}{\thesection}{1em}{}
\titleformat{\subsection}{\normalfont\normalsize\bfseries}{\thesubsection}{1em}{}
\titleformat{\subsubsection}{\normalfont\normalsize\bfseries}{\thesubsubsection}{1em}{}

% Numbering depth: numbered down to 5 levels (e.g. 5.1.1.1.1)
\setcounter{secnumdepth}{5}
\setcounter{tocdepth}{5}

% Enable \paragraph / \subparagraph as numbered 4th/5th level headings
\titleformat{\paragraph}[hang]
  {\normalfont\normalsize\bfseries}{\theparagraph}{1em}{}
\titlespacing*{\paragraph}{0pt}{2.0ex plus 1ex minus .2ex}{1.0ex plus .2ex}
\titleformat{\subparagraph}[hang]
  {\normalfont\normalsize\bfseries}{\thesubparagraph}{1em}{}
\titlespacing*{\subparagraph}{0pt}{2.0ex plus 1ex minus .2ex}{1.0ex plus .2ex}

}

\newcommand{\frontchapter}[1]{%
  \chapter*{#1}%
  \addcontentsline{toc}{chapter}{#1}%
  \markboth{#1}{#1}%
}
